function calculateMatrixParity(matrix, dimensions) {
    let absoluteError = 0.0;
    const cacheStride = dimensions.width * 4;
    for (let i = 0; i < dimensions.height; i++) {
        for (let j = 0; j < dimensions.width; j++) {
            const index = i * cacheStride + j;
            absoluteError += Math.abs(matrix[index]);
        }
    }
    return absoluteError;
}
