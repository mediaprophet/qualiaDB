Native Scientific Primitives

QualiaDB includes zero-allocation scientific computing engines wired directly into the Webizen VM:

MCMC Metropolis-Hastings, Boltzmann acceptance, Gibbs free energy

Runge-Kutta 4th-order integrator for differential equations

SMILES/InChI parsing, Lipinski filters, Morgan fingerprint

Zero-Heap Execution

All scientific primitives are zero-allocation. They use stack-allocated buffers and caller-supplied output arrays, ensuring deterministic performance even on constrained devices.

Thermodynamics

MCMC Metropolis-Hastings

# Sample from a Boltzmann distribution
SELECT ?energy ?probability WHERE {
  ?state a ex:ThermodynamicState ;
         ex:energy ?energy .
  
  # Native thermodynamics primitive
  BIND(qualia:mcmc_boltzmann(?energy, 300.0) AS ?probability)
}

Gibbs Free Energy

# Calculate Gibbs free energy
SELECT ?reaction ?deltaG WHERE {
  ?reaction a ex:ChemicalReaction ;
            ex:enthalpy ?H ;
            ex:entropy ?S ;
            ex:temperature ?T .
  
  # ΔG = ΔH - TΔS
  BIND(qualia:gibbs_free_energy(?H, ?S, ?T) AS ?deltaG)
}

Phase Transition Detection

# Detect phase transitions in material
SELECT ?temperature ?phase WHERE {
  ?material a ex:Material ;
            ex:temperature ?temperature ;
            ex:heat_capacity ?Cp .
  
  # Phase transition when heat capacity spikes
  BIND(qualia:detect_phase_transition(?Cp) AS ?phase)
}

ODE Solver

Runge-Kutta 4th Order

# Solve a differential equation
SELECT ?time ?value WHERE {
  # Initial conditions
  BIND(0.0 AS ?t0)
  BIND(1.0 AS ?y0)
  
  # Solve dy/dt = -2y over time [0, 5]
  BIND(qualia:runge_kutta_4(?t0, ?y0, 5.0, 0.1, "-2*y") AS ?trajectory)
  
  # Extract time and value from trajectory
  BIND(qualia:trajectory_time(?trajectory, ?i) AS ?time)
  BIND(qualia:trajectory_value(?trajectory, ?i) AS ?value)
}

Population Dynamics

# Lotka-Volterra predator-prey model
SELECT ?time ?prey ?predator WHERE {
  # dx/dt = αx - βxy
  # dy/dt = δxy - γy
  
  BIND(qualia:runge_kutta_4_system(
    0.0,  # t0
    [10.0, 5.0],  # [x0, y0] initial prey/predator
    50.0,  # t_max
    0.1,  # dt
    "1.5*x - 0.5*x*y",  # dx/dt
    "0.5*x*y - 1.0*y"   # dy/dt
  ) AS ?solution)
}

Epidemic Modeling (SIR)

# SIR epidemic model
SELECT ?time ?susceptible ?infected ?recovered WHERE {
  BIND(qualia:sir_model(
    0.0,      # t0
    990,      # S0 (susceptible)
    10,       # I0 (infected)
    0,        # R0 (recovered)
    100,      # t_max
    0.1,      # dt
    0.3,      # β (infection rate)
    0.1       # γ (recovery rate)
  ) AS ?trajectory)
}

Bioinformatics

Smith-Waterman Alignment

# Align two DNA sequences
SELECT ?score ?alignment WHERE {
  BIND("ACGTACGT" AS ?seq1)
  BIND("ACGTTACGT" AS ?seq2)
  
  # Smith-Waterman local alignment
  BIND(qualia:smith_waterman(?seq1, ?seq2) AS ?result)
  BIND(qualia:alignment_score(?result) AS ?score)
  BIND(qualia:alignment_string(?result) AS ?alignment)
}

DNA to Protein Translation

# Translate DNA to protein
SELECT ?dna ?protein WHERE {
  BIND("ATGGCCATTGTA" AS ?dna)
  
  # Translate using standard genetic code
  BIND(qualia:dna_to_protein(?dna) AS ?protein)
}
# Result: "MAIV"

K-mer Hashing

# Generate k-mer hashes for sequence comparison
SELECT ?kmer ?hash WHERE {
  BIND("ACGTACGT" AS ?sequence)
  BIND(3 AS ?k)
  
  # Generate all k-mers and their hashes
  BIND(qualia:kmer_hashes(?sequence, ?k) AS ?kmer_hashes)
}

Clinical Risk Assessment

Framingham Risk Score

# Calculate 10-year cardiovascular risk
SELECT ?patient ?risk_score WHERE {
  ?patient a ex:Patient ;
           ex:age 55 ;
           ex:gender "male" ;
           ex:total_cholesterol 220 ;
           ex:hdl_cholesterol 45 ;
           ex:systolic_bp 140 ;
           ex:smoker true ;
           ex:treated_for_bp false .
  
  BIND(qualia:framingham_risk(
    ?age, ?gender, ?total_cholesterol, ?hdl_cholesterol,
    ?systolic_bp, ?smoker, ?treated_for_bp
  ) AS ?risk_score)
}

Drug Interaction Check

# Check for drug-drug interactions
SELECT ?patient ?drug1 ?drug2 ?severity WHERE {
  ?patient ex:takes ?drug1, ?drug2 .
  
  BIND(qualia:drug_interaction(?drug1, ?drug2) AS ?interaction)
  BIND(qualia:interaction_severity(?interaction) AS ?severity)
  FILTER (?severity > 0)  # Only show interactions
}

Organic Chemistry

SMILES Parsing

# Parse SMILES notation
SELECT ?smiles ?molecular_weight WHERE {
  BIND("CCO" AS ?smiles)  # Ethanol
  
  BIND(qualia:parse_smiles(?smiles) AS ?molecule)
  BIND(qualia:molecular_weight(?molecule) AS ?molecular_weight)
}
# Result: 46.07 g/mol

Lipinski Rule of Five

# Check drug-likeness using Lipinski rules
SELECT ?molecule ?passes_lipinski WHERE {
  ?molecule a ex:Molecule ;
            ex:smiles "CC(C)CC1=CC=C(C=C1)C(C)C(=O)O" ;  # Ibuprofen
            ex:molecular_weight ?mw ;
            ex:logp ?logp ;
            ex:hbd ?hbd ;  # Hydrogen bond donors
            ex:hba ?hba .  # Hydrogen bond acceptors
  
  BIND(qualia:lipinski_rule_of_five(?mw, ?logp, ?hbd, ?hba) AS ?passes_lipinski)
}

Morgan Fingerprint

# Generate Morgan fingerprint for similarity search
SELECT ?molecule ?fingerprint WHERE {
  ?molecule a ex:Molecule ;
            ex:smiles "CCO" .
  
  BIND(qualia:morgan_fingerprint(?molecule, 2, 2048) AS ?fingerprint)
}
