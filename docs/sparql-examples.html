SPARQL Query Examples

Comprehensive examples of SPARQL queries in QualiaDB, including extensions

Basic SPARQL Queries

SELECT: Simple Triple Pattern

SELECT ?s ?p ?o WHERE {
  ?s ?p ?o
} LIMIT 10

SELECT: Filter by Type

SELECT ?person ?name WHERE {
  ?person a ex:Person ;
           ex:name ?name .
}

SELECT: Multiple Patterns

SELECT ?person ?name ?age WHERE {
  ?person a ex:Person ;
           ex:name ?name ;
           ex:age ?age .
  FILTER (?age > 30)
}

ASK: Existence Check

ASK {
  ?person ex:name "Alice" .
}

Property Paths

QualiaDB supports SPARQL 1.1 property paths:

Sequence Path (/)

# Find friends of friends
SELECT ?person ?friendOfFriend WHERE {
  ?person ex:knows/ex:knows ?friendOfFriend .
}

Inverse Path (^)

# Find who knows Alice
SELECT ?knower WHERE {
  ?knower ^ex:knows ex:alice .
}

Alternative Path (|)

# Find people connected by any relationship
SELECT ?person ?related WHERE {
  ?person (ex:knows|ex:worksWith|ex:relatedTo) ?related .
}

Zero or More (*)

# Find all descendants in a hierarchy
SELECT ?ancestor ?descendant WHERE {
  ?ancestor ex:parent* ?descendant .
}

SPARQL-Star (RDF-Star)

QualiaDB supports SPARQL-Star for annotating triples:

Nested Triple Pattern

# Find triples with confidence scores
SELECT ?s ?p ?o ?confidence WHERE {
  << ?s ?p ?o >> ex:confidence ?confidence .
}

Annotating Query Results

# Return annotated triples
SELECT ?triple ?source WHERE {
  ?triple a ex:Statement ;
          ex:source ?source .
}

Filtering on Annotations

# Find high-confidence relationships
SELECT ?s ?p ?o WHERE {
  << ?s ?p ?o >> ex:confidence ?conf .
  FILTER (?conf > 0.9)
}

Aggregates

Standard SPARQL 1.1 aggregates:

COUNT

SELECT (COUNT(?person) AS ?count) WHERE {
  ?person a ex:Person .
}

GROUP BY

SELECT ?city (COUNT(?person) AS ?population) WHERE {
  ?person a ex:Person ;
           ex:city ?city .
} GROUP BY ?city

AVG, MIN, MAX, SUM

SELECT (AVG(?age) AS ?avgAge)
       (MIN(?age) AS ?minAge)
       (MAX(?age) AS ?maxAge)
       (SUM(?salary) AS ?totalSalary) WHERE {
  ?person ex:age ?age ;
           ex:salary ?salary .
}

HAVING

SELECT ?city (COUNT(?person) AS ?population) WHERE {
  ?person a ex:Person ;
           ex:city ?city .
} GROUP BY ?city
HAVING (?population > 1000)

QualiaDB Extensions

QualiaDB-specific SPARQL extensions:

AS OF (Temporal Query)

# Query data as of a specific date
SELECT ?person ?name WHERE {
  ?person ex:name ?name .
} AS OF "2024-01-01"^^xsd:dateTime

AT TIME (Millisecond Timestamp)

# Query data at a specific millisecond timestamp
SELECT ?person ?name WHERE {
  ?person ex:name ?name .
} AT TIME 1704067200000

SERVICE (Federated Query)

# Query remote SPARQL endpoint
SELECT ?s ?p ?o WHERE {
  SERVICE  {
    ?s ?p ?o .
    FILTER (?s = )
  }
}

GeoSPARQL (Spatial)

# Find locations within a radius
SELECT ?location ?name WHERE {
  ?location a ex:Location ;
            ex:name ?name ;
            geo:asWKT ?wkt .
  FILTER (geof:distance(?wkt, "POINT(-122.4194 37.7749)"^^geo:wktLiteral) < 10.0)
}

SPARQL UPDATE

SPARQL 1.1 UPDATE operations:

INSERT DATA

INSERT DATA {
  ex:alice a ex:Person ;
           ex:name "Alice Smith" ;
           ex:age 30 .
}

DELETE DATA

DELETE DATA {
  ex:alice ex:age 30 .
}

DELETE WHERE

DELETE WHERE {
  ?person ex:age ?age .
  FILTER (?age < 18)
}

INSERT WHERE

INSERT {
  ?person ex:adult true .
}
WHERE {
  ?person ex:age ?age .
  FILTER (?age >= 18)
}

DELETE INSERT

DELETE {
  ?person ex:age ?oldAge .
}
INSERT {
  ?person ex:age ?newAge .
}
WHERE {
  ?person ex:age ?oldAge .
  BIND (?oldAge + 1 AS ?newAge)
}

SHACL Integration

QualiaDB integrates SHACL validation with SPARQL:

SHACL-SPARQL Constraints

ex:PersonShape a sh:NodeShape ;
  sh:targetClass ex:Person ;
  sh:property [
    sh:path ex:age ;
    sh:datatype xsd:integer ;
    sh:minInclusive 0 ;
    sh:maxInclusive 150 ;
    sh:sparql [
      sh:select """
        SELECT $this ($this AS ?value)
        WHERE {
          FILTER NOT EXISTS {
            $this ex:age ?age .
            FILTER (?age >= 18)
          }
        }
      """ ;
    ]
  ] .

Validate with SPARQL

# Validate data against SHACL shapes
SELECT ?node ?constraint ?message WHERE {
  ?node a ex:Person .
  # SHACL validation logic
}
