let p_ndc = vec4<f32>(uv * 2.0 - 1.0, depth, 1.0); let p_world = inv_view_proj * p_ndc; let p_prev = prev_view_proj * (p_world / p_world.w); let motion = ((p_ndc.xy - p_prev.xy / p_prev.w) * 0.5);

// WGSL GPU Temporal Warp & Reprojection Kernel @compute @workgroup_size(64, 1, 1) fn cs_temporal_warp(@builtin(global_invocation_id) gid: vec3<u32>) { let idx = gid.x; if (idx >= uniforms.vertex_count) { return; } let curr_pos = curr_positions[idx].xyz; let prev_clip = prev_view_proj * vec4<f32>(curr_pos, 1.0); let prev_uv = (prev_clip.xy / prev_clip.w) * vec2<f32>(0.5, -0.5) + vec2<f32>(0.5, 0.5); motion_vectors[idx] = vec4<f32>(curr_pos - prev_positions[idx].xyz, prev_uv.x); }
